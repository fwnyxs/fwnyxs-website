set pagesize 1000
set linesize 220
set trimspool on
set verify off
set feedback off

-- 1. FORMAT UNTUK DATAFILE & TEMPFILE
col Tablespace for a25
col Filename for a70 wrapped
col autoextensible for a6 heading "Auto|Extend"
col MaxSizeMB for 999,999,999 heading "Max Size|(MB)"
col UsedSizeMB for 999,999,999 heading "Curr Size|(MB)"
col Space_Left_MB for 999,999,999 heading "Max Free|Left (MB)"

-- 2. FORMAT UNTUK SUMMARY TABLESPACE (BAHAGIAN BAWAH)
col "Tablepace_name" for a25
col "AutExtent" for a6
col "NoOfFile" heading "No of|Files" for 999
col "MaxTblSpaceSize(MB)" heading "Tablespace|Max Size(MB)" for 999,999,999
col "CurrTblSpaceSize(MB)" heading "Tablespace|Current Size(MB)" for 999,999,999
col "UsedSpace(MB)" heading "Tablespace|Used Space(MB)" for 999,999,999
col "CurrFreeSpace(MB)" heading "Tablespace Current|Free Space (MB)" for 999,999,999
col "TotFreeSpace(MB)" heading "Tablespace Total|Free Space (MB)" for 999,999,999
col "Used(%)" for 999.99
col "Free(%)" for 999.99

pro
pro ================================================================================
pro ************************** 1. DATAFILE USAGE **********************************
pro ================================================================================

break on Tablespace skip 1
select tablespace_name "Tablespace",
       file_name "Filename",
       autoextensible,
       round(maxbytes/1024/1024) "MaxSizeMB",
       round(user_bytes/1024/1024) "UsedSizeMB",
       round((maxbytes-bytes)/1024/1024) "Space_Left_MB" 
from dba_data_files 
order by tablespace_name, file_name;

pro
pro ================================================================================
pro ************************** 2. TEMPFILE USAGE **********************************
pro ================================================================================

select tablespace_name "Tablespace",
       file_name "Filename",
       autoextensible,
       round(maxbytes/1024/1024) "MaxSizeMB",
       round(user_bytes/1024/1024) "UsedSizeMB",
       round((maxbytes-bytes)/1024/1024) "Space_Left_MB" 
from dba_temp_files 
order by tablespace_name, file_name;

pro
pro ================================================================================
pro *************** 3. SUMMARY TABLESPACE USAGE (PERMANENT & UNDO) *****************
pro ================================================================================

WITH tbs_auto AS
     (SELECT DISTINCT tablespace_name, autoextensible
                 FROM dba_data_files
                WHERE autoextensible = 'YES'
      UNION 
      SELECT DISTINCT tablespace_name, autoextensible 
                 FROM dba_data_files 
                WHERE tablespace_name NOT IN (SELECT tablespace_name FROM dba_data_files WHERE autoextensible='YES')),
     files AS
     (SELECT   tablespace_name, COUNT (*) tbs_files,
               SUM (BYTES) total_tbs_bytes
          FROM dba_data_files
      GROUP BY tablespace_name),
     fragments AS
     (SELECT   tablespace_name, 
               SUM (BYTES) total_tbs_free_bytes
          FROM dba_free_space
      GROUP BY tablespace_name),
     AUTOEXTEND AS
     (SELECT   TableSpace_Name , SUM (size_to_grow) total_growth_tbs
          FROM (SELECT   tablespace_name, SUM (maxbytes) size_to_grow
                    FROM dba_data_files
                   WHERE autoextensible = 'YES'
                GROUP BY tablespace_name
                UNION
                SELECT   tablespace_name, SUM (BYTES) size_to_grow
                    FROM dba_data_files
                   WHERE autoextensible = 'NO'
                GROUP BY tablespace_name)
      GROUP BY tablespace_name)
SELECT a.tablespace_name "Tablepace_name",
       nvl(tbs_auto.autoextensible, 'NO') "AutExtent",
       files.tbs_files "NoOfFile",
       round(AUTOEXTEND.total_growth_tbs/1024/1024, 2) "MaxTblSpaceSize(MB)",
       round(files.total_tbs_bytes/1024/1024, 2) "CurrTblSpaceSize(MB)",
       round((files.total_tbs_bytes - nvl(fragments.total_tbs_free_bytes,0))/1024/1024, 2) "UsedSpace(MB)",
       round(nvl(fragments.total_tbs_free_bytes,0)/1024/1024, 2) "CurrFreeSpace(MB)",
       round((AUTOEXTEND.total_growth_tbs - (files.total_tbs_bytes - nvl(fragments.total_tbs_free_bytes,0)))/1024/1024, 2) "TotFreeSpace(MB)",
       round((( (files.total_tbs_bytes - nvl(fragments.total_tbs_free_bytes,0))/ decode(AUTOEXTEND.total_growth_tbs,0,1,AUTOEXTEND.total_growth_tbs))* 100),2) "Used(%)",
       100 - round((( (files.total_tbs_bytes - nvl(fragments.total_tbs_free_bytes,0))/ decode(AUTOEXTEND.total_growth_tbs,0,1,AUTOEXTEND.total_growth_tbs))* 100),2) "Free(%)"
  FROM dba_tablespaces a, files, fragments, AUTOEXTEND, tbs_auto
 WHERE a.tablespace_name = files.tablespace_name(+)
   AND a.tablespace_name = fragments.tablespace_name(+)
   AND a.tablespace_name = AUTOEXTEND.tablespace_name(+)
   AND a.tablespace_name = tbs_auto.tablespace_name(+)
   AND a.contents IN ('PERMANENT', 'UNDO')
 order by 1;
